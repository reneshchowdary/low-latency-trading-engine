cmake_minimum_required(VERSION 3.20)
project(low_latency_trading_engine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(trading_engine src/order_book.cpp)
target_include_directories(trading_engine PUBLIC include)
target_compile_options(trading_engine PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
add_executable(trading_demo src/main.cpp)
target_link_libraries(trading_demo PRIVATE trading_engine)
add_executable(matching_benchmark benchmarks/matching_benchmark.cpp)
target_link_libraries(matching_benchmark PRIVATE trading_engine)
enable_testing()
add_executable(order_book_tests tests/order_book_tests.cpp)
target_link_libraries(order_book_tests PRIVATE trading_engine)
add_test(NAME order_book_tests COMMAND order_book_tests)
