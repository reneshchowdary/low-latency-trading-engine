cmake_minimum_required(VERSION 3.20)
project(low_latency_trading_engine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Enable ASan and UBSan" OFF)
find_package(Threads REQUIRED)
add_library(trading_engine src/order_book.cpp src/protocol.cpp)
target_include_directories(trading_engine PUBLIC include)
target_compile_options(trading_engine PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
if(ENABLE_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
  target_compile_options(trading_engine PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(trading_engine PUBLIC -fsanitize=address,undefined)
endif()
add_executable(trading_demo src/main.cpp)
target_link_libraries(trading_demo PRIVATE trading_engine)
add_executable(trading_gateway src/tcp_gateway.cpp)
target_link_libraries(trading_gateway PRIVATE trading_engine)
add_executable(matching_benchmark benchmarks/matching_benchmark.cpp)
target_link_libraries(matching_benchmark PRIVATE trading_engine)
enable_testing()
add_executable(order_book_tests tests/order_book_tests.cpp)
target_link_libraries(order_book_tests PRIVATE trading_engine)
add_test(NAME order_book_tests COMMAND order_book_tests)
add_executable(protocol_queue_tests tests/protocol_queue_tests.cpp)
target_link_libraries(protocol_queue_tests PRIVATE trading_engine Threads::Threads)
add_test(NAME protocol_queue_tests COMMAND protocol_queue_tests)

find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_test(NAME tcp_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/tcp_integration.py $<TARGET_FILE:trading_gateway>)
endif()

add_executable(risk_tests tests/risk_tests.cpp)
target_link_libraries(risk_tests PRIVATE trading_engine)
add_test(NAME risk_tests COMMAND risk_tests)
